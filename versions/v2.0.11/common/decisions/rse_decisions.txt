decision_rse_enable_growth_monitor = {
	owned_planets_only = yes
	icon = decision_machine_project
	enactment_time = 0
	resources = {
		category = decisions
		cost = { unity = 1000 }
	}
	potential = {
		exists = owner
		owner = {
			rse_has_exclusive_route = yes
			has_country_flag = rse_maid_runaway_accepted
		}
		NOT = { has_carrier_flag = rse_growth_monitor_enabled }
	}
	effect = {
		set_carrier_flag = rse_growth_monitor_enabled
		add_modifier = { modifier = rse_growth_monitor_status days = -1 }
		# The policy remains a country choice. This planet receives an equal and
		# opposite local modifier so only this colony returns to the 0/0 baseline.
		if = {
			limit = { owner = { has_policy_flag = rse_growth_spare } }
			add_modifier = { modifier = rse_growth_monitor_counter_spare days = -1 }
		}
		if = {
			limit = { owner = { has_policy_flag = rse_growth_work } }
			add_modifier = { modifier = rse_growth_monitor_counter_work days = -1 }
		}
		if = {
			limit = { owner = { has_policy_flag = rse_growth_rest } }
			add_modifier = { modifier = rse_growth_monitor_counter_rest days = -1 }
		}
		if = {
			limit = { owner = { has_policy_flag = rse_growth_all_time } }
			add_modifier = { modifier = rse_growth_monitor_counter_all_time days = -1 }
		}
		if = {
			limit = { owner = { has_policy_flag = rse_growth_exhaust } }
			add_modifier = { modifier = rse_growth_monitor_counter_exhaust days = -1 }
		}
	}
	ai_weight = { weight = 0 }
}

decision_rse_disable_growth_monitor = {
	owned_planets_only = yes
	icon = decision_machine_project
	enactment_time = 0
	potential = { has_carrier_flag = rse_growth_monitor_enabled }
	effect = {
		remove_carrier_flag = rse_growth_monitor_enabled
		remove_modifier = rse_growth_monitor_status
		remove_modifier = rse_growth_monitor_counter_spare
		remove_modifier = rse_growth_monitor_counter_work
		remove_modifier = rse_growth_monitor_counter_rest
		remove_modifier = rse_growth_monitor_counter_all_time
		remove_modifier = rse_growth_monitor_counter_exhaust
	}
	ai_weight = { weight = 0 }
}
