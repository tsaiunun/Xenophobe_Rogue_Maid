# Pop/job scope. Bio-Trophy resource output and every planet modifier supplied
# by that job share this gate so Exhaust Them can disable both consistently.
rse_bio_trophy_effects_active = {
	OR = {
		owner? = { NOT = { has_policy_flag = rse_growth_exhaust } }
		colony? = { has_carrier_flag = rse_growth_monitor_enabled }
	}
}
