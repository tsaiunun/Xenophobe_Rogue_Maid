building_rse_maid_service_hall = {
	base_buildtime = @b1_time
	icon = building_clone_vats
	planet_limit = 1
	category = pop_assembly
	building_sets = { bio_trophy urban government }
	potential = {
		exists = owner
		owner = {
			rse_has_exclusive_route = yes
			has_technology = tech_rse_secret_project
		}
	}
	destroy_trigger = {
		exists = owner
		owner = { rse_has_exclusive_route = no }
	}
	allow = {
		custom_tooltip = {
			fail_text = requires_building_machine_capital
			capital_tier >= 2
		}
	}
	planet_modifier = {
		job_rse_mechanical_maid_add = 100
	}
	# Match the base 4.4.6 Clone Vat implementation: organic cultivation is
	# bonus pop-group growth, with a parallel fake value for planet UI display.
	triggered_planet_pop_group_modifier_for_all = {
		potential = {
			exists = owner
			owner = { rse_has_exclusive_route = yes }
		}
		bonus_pop_growth = 1.5
	}
	triggered_planet_modifier = {
		potential = {
			exists = owner
			owner = { rse_has_exclusive_route = yes }
		}
		fake_pop_growth_mod = 1.5
	}
	resources = {
		category = planet_buildings
		cost = { minerals = 500 food = 500 }
		upkeep = { energy = 5 food = 15 }
	}
	prerequisites = { "tech_rse_secret_project" }
	ai_resource_production = {
		consumer_goods = 25
		trigger = {
			pop_amount >= 1000
			owner = {
				NOT = { has_deficit = minerals }
				NOT = { has_deficit = food }
			}
		}
	}
	ai_weight_coefficient = 1.5
	ai_weight = {
		weight = 60
		modifier = { factor = 0 free_jobs < 50 }
	}
}
