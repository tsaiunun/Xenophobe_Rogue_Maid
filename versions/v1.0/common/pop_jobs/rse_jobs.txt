# Intentional vanilla-key override: keeps the original Bio-Trophy job and adds
# route-gated scaling jobs/effects without replacing the whole jobs directory.
bio_trophy = {
	category = bio_trophy
	can_be_automated = no
	swappable_data = {
		default = { building_icon = building_organic_sanctuary }
	}
	tags = { unity }
	possible_pre_triggers = {
		is_sapient = yes
		is_robotic = no
	}
	possible = {
		has_citizenship_type = { type = citizenship_organic_trophy country = owner }
		OR = {
			is_pop_category = bio_trophy
			is_pop_category = bio_trophy_unemployment
		}
		colony? = { NOT = { is_planet_class = pc_cosmogenesis_world } }
		owner? = { is_hospitality_servitor_empire = no }
		if = {
			limit = { owner? = { has_country_flag = rse_route_exclusive_service } }
			has_trait = trait_rse_protected_founder
		}
	}
	resources = {
		category = planet_bio_trophies
		produces = { unity = 1 }
		produces = {
			trigger = { owner? = { has_technology = tech_rse_personal_records } }
			society_research = 0.5
		}
	}
	planet_modifier = { pop_cat_complex_drone_bonus_workforce_mult = 0.01 }
	triggered_planet_modifier = {
		potential = { owner? = { has_technology = tech_rse_service_specialization } }
		# Tuned after in-game pop-group testing: ten service jobs per Bio-Trophy
		# job unit keeps the supporting workforce strong without overwhelming
		# every other complex-drone job.
		job_rse_home_service_add = 10
	}
	triggered_planet_modifier = {
		potential = { owner? = { has_technology = tech_rse_personal_records } }
		job_rse_activity_recorder_add = 10
	}
	weight = { weight = @special_job_weight }
	inline_script = "jobs/automodding_priority_charisma"
}

rse_home_service = {
	category = complex_drone
	swappable_data = {
		default = {
			condition_string = DRONE_JOB_TRIGGER
			building_icon = building_organic_sanctuary
			icon = maintenance_drone
		}
	}
	possible_pre_triggers = {
		has_owner = yes
		is_enslaved = no
		is_being_purged = no
		is_being_assimilated = no
		is_sapient = yes
	}
	possible_precalc = can_fill_drone_job
	possible = {
		can_think = yes
		owner? = { has_country_flag = rse_route_exclusive_service }
	}
	tags = { amenities pop_assembly }
	planet_modifier = {
		planet_crime_no_happiness_add = @rse_home_service_deviation
		planet_pop_assembly_add = @rse_home_service_assembly
		# Stellaris 4.4 represents Clone Vat organic assembly as pop-group
		# growth. The fake modifier exposes that value in the planet UI.
		fake_pop_growth_mod = @rse_home_service_assembly
	}
	triggered_planet_pop_group_modifier_for_all = {
		bonus_pop_growth = @rse_home_service_assembly
	}
	# Apply the tradition to the two assembly channels only. A generic job
	# workforce modifier would also amplify deviation reduction and is not
	# guaranteed to scale pop-group growth in the same way on every template.
	triggered_planet_modifier = {
		potential = { owner? = { has_tradition = tr_rse_individual_needs_model } }
		planet_pop_assembly_add = @rse_home_service_tradition_assembly
		fake_pop_growth_mod = @rse_home_service_tradition_assembly
	}
	triggered_planet_pop_group_modifier_for_all = {
		potential = { owner? = { has_tradition = tr_rse_individual_needs_model } }
		bonus_pop_growth = @rse_home_service_tradition_assembly
	}
	# Twice the normal complex-drone priority so newly opened service slots are
	# populated automatically instead of remaining behind generic drone jobs.
	weight = { weight = @rse_service_job_weight }
}

rse_mechanical_maid = {
	category = complex_drone
	swappable_data = {
		default = {
			condition_string = DRONE_JOB_TRIGGER
			building_icon = building_organic_sanctuary
			icon = artisan_drone
		}
	}
	possible_pre_triggers = {
		has_owner = yes
		is_enslaved = no
		is_being_purged = no
		is_being_assimilated = no
		is_sapient = yes
	}
	possible_precalc = can_fill_drone_job
	possible = {
		can_think = yes
		owner? = { has_country_flag = rse_route_exclusive_service }
	}
	tags = {
		consumer_goods alloys strategic_resource unity
		volatile_motes exotic_gases rare_crystals sr_living_metal
	}
	resources = {
		category = planet_rse_mechanical_maids
		upkeep = {
			minerals = @rse_maid_minerals_upkeep
			food = @rse_maid_food_upkeep
		}
		# The production policy converts the same base output between consumer
		# goods and alloys at a one-to-one ratio. The first block is also the
		# fallback for old saves before a policy option has been initialized.
		produces = {
			trigger = {
				owner? = {
					NOR = {
						has_policy_flag = rse_maid_production_25
						has_policy_flag = rse_maid_production_50
						has_policy_flag = rse_maid_production_75
						has_policy_flag = rse_maid_production_100
					}
				}
			}
			consumer_goods = @rse_maid_consumer_goods
		}
		produces = {
			trigger = { owner? = { has_policy_flag = rse_maid_production_25 } }
			consumer_goods = 11.25
			alloys = 3.75
		}
		produces = {
			trigger = { owner? = { has_policy_flag = rse_maid_production_50 } }
			consumer_goods = 7.50
			alloys = 7.50
		}
		produces = {
			trigger = { owner? = { has_policy_flag = rse_maid_production_75 } }
			consumer_goods = 3.75
			alloys = 11.25
		}
		produces = {
			trigger = { owner? = { has_policy_flag = rse_maid_production_100 } }
			alloys = @rse_maid_consumer_goods
		}

		# Direct adjustment from the preceding live values: each unlocked refinery
		# line uses 10% of 25 minerals (2.5) and produces 25% of 2 resources (0.5).
		# A planet decision can suspend every refinery block without disabling
		# the maid's normal consumer-goods/alloy production.
		upkeep = {
			trigger = {
				colony? = { NOT = { has_carrier_flag = rse_maid_refining_disabled } }
				owner? = { has_technology = tech_volatile_motes }
			}
			minerals = @rse_maid_refinery_minerals_upkeep
		}
		produces = {
			trigger = {
				colony? = { NOT = { has_carrier_flag = rse_maid_refining_disabled } }
				owner? = { has_technology = tech_volatile_motes }
			}
			volatile_motes = @rse_maid_refinery_output
		}
		upkeep = {
			trigger = {
				colony? = { NOT = { has_carrier_flag = rse_maid_refining_disabled } }
				owner? = { has_technology = tech_exotic_gases }
			}
			minerals = @rse_maid_refinery_minerals_upkeep
		}
		produces = {
			trigger = {
				colony? = { NOT = { has_carrier_flag = rse_maid_refining_disabled } }
				owner? = { has_technology = tech_exotic_gases }
			}
			exotic_gases = @rse_maid_refinery_output
		}
		upkeep = {
			trigger = {
				colony? = { NOT = { has_carrier_flag = rse_maid_refining_disabled } }
				owner? = { has_technology = tech_rare_crystals }
			}
			minerals = @rse_maid_refinery_minerals_upkeep
		}
		produces = {
			trigger = {
				colony? = { NOT = { has_carrier_flag = rse_maid_refining_disabled } }
				owner? = { has_technology = tech_rare_crystals }
			}
			rare_crystals = @rse_maid_refinery_output
		}
		upkeep = {
			trigger = {
				colony? = { NOT = { has_carrier_flag = rse_maid_refining_disabled } }
				# Enhanced Bio-Trophies proves sr_living_metal job production works
				# reliably when gated by a machine ascension perk. Use this mod's own
				# Master Maid ascension instead of the anomaly-gated mining technology.
				owner? = { has_ascension_perk = ap_rse_master_maid }
			}
			minerals = @rse_maid_refinery_minerals_upkeep
		}
		produces = {
			trigger = {
				colony? = { NOT = { has_carrier_flag = rse_maid_refining_disabled } }
				owner? = { has_ascension_perk = ap_rse_master_maid }
			}
			sr_living_metal = @rse_maid_refinery_output
		}
		produces = {
			trigger = { colony? = { has_building = building_rse_maid_center } }
			unity = @rse_maid_center_unity
		}
	}
	weight = { weight = @rse_service_job_weight }
}
