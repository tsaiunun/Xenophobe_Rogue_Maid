# A neutral living standard paired with citizenship_rse_general_organic. It
# deliberately creates ordinary workers/specialists rather than Bio-Trophies.
living_standard_rse_general_organic = {
	triggered_pop_group_modifier = {
		potential = {
			owner? = { NOT = { has_deficit = consumer_goods } }
		}
		pop_happiness = 0.10
	}
	triggered_pop_group_modifier = {
		potential = {
			owner? = { has_deficit = consumer_goods }
		}
		pop_happiness = -0.10
	}

	pop_group_modifier = {
		pop_category_rulers_consumer_goods_upkeep_add = 1
		pop_category_specialists_consumer_goods_upkeep_add = 1
		pop_category_workers_consumer_goods_upkeep_add = 0.5
		pop_category_civilians_consumer_goods_upkeep_add = 0.5
	}

	potential = {
		exists = from
		from = { rse_has_exclusive_route = yes }
		NOR = {
			has_trait = trait_mechanical
			has_trait = trait_machine_unit
			has_trait = trait_rse_protected_founder
		}
	}

	allow = {
		hidden_trigger = { exists = from }
		custom_tooltip = {
			fail_text = rse_requires_general_organic_citizenship
			has_citizenship_type = {
				country = from
				type = citizenship_rse_general_organic
			}
		}
	}

	ai_will_do = { factor = 1 }
}
