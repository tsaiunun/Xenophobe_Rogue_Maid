# Intentional vanilla-key override: keeps the original Bio-Trophy job and adds
# route-gated scaling jobs/effects without replacing the whole jobs directory.
bio_trophy = {
	category = bio_trophy
	can_be_automated = no
	swappable_data = {
		default = { building_icon = building_organic_sanctuary }
	}
	tags = { unity }
	possible_pre_triggers = {
		is_sapient = yes
		is_robotic = no
	}
	possible = {
		has_citizenship_type = { type = citizenship_organic_trophy country = owner }
		OR = {
			is_pop_category = bio_trophy
			is_pop_category = bio_trophy_unemployment
		}
		colony? = { NOT = { is_planet_class = pc_cosmogenesis_world } }
		owner? = { is_hospitality_servitor_empire = no }
		if = {
			limit = { owner? = { has_country_flag = rse_route_exclusive_service } }
			has_trait = trait_rse_protected_founder
			rse_bio_trophy_effects_active = yes
		}
	}
	resources = {
		category = planet_bio_trophies
		# "Exhaust Them" keeps the existing job slots for save compatibility but
		# suppresses all output unless the local growth monitor is enabled.
		produces = {
			trigger = { rse_bio_trophy_effects_active = yes }
			unity = 1
		}
		produces = {
			trigger = {
				owner? = { has_technology = tech_rse_service_specialization }
				rse_bio_trophy_effects_active = yes
			}
			unity = 1
		}
		produces = {
			trigger = {
				owner? = { has_technology = tech_rse_personal_records }
				rse_bio_trophy_effects_active = yes
			}
			unity = 1
			society_research = 0.5
		}
		produces = {
			trigger = {
				owner? = { has_technology = tech_rse_personal_records }
				rse_bio_trophy_effects_active = yes
			}
			physics_research = @rse_bio_trophy_science
			society_research = @rse_bio_trophy_science
			engineering_research = @rse_bio_trophy_science
		}
		produces = {
			trigger = {
				colony? = { has_modifier = rse_white_leader_common_sense }
				rse_bio_trophy_effects_active = yes
			}
			food = 1
		}
	}
	triggered_planet_modifier = {
		potential = {
			rse_bio_trophy_effects_active = yes
			owner? = {
				OR = {
					NOT = { has_technology = tech_rse_drone_rotation }
					has_policy_flag = rse_activity_scope_urban
				}
			}
		}
		pop_cat_complex_drone_bonus_workforce_mult = 0.01
	}
	triggered_planet_modifier = {
		potential = {
			rse_bio_trophy_effects_active = yes
			owner? = {
				has_ascension_perk = ap_rse_master_maid
			}
		}
		# Add 0.3 percentage points to each activity-policy baseline.
		pop_cat_complex_drone_bonus_workforce_mult = 0.003
		worker_and_simple_drone_cat_bonus_workforce_mult = 0.003
	}
	triggered_country_modifier = {
		potential = {
			rse_bio_trophy_effects_active = yes
			owner? = {
				any_owned_starbase = { has_starbase_building = rse_custom_program_center }
			}
		}
		country_base_influence_produces_add = @rse_bio_trophy_influence
	}
	triggered_planet_modifier = {
		potential = {
			rse_bio_trophy_effects_active = yes
			owner? = {
				any_owned_starbase = { has_starbase_building = rse_planet_tv_center }
			}
		}
		planet_building_build_speed_mult = @rse_bio_trophy_build_speed
	}
	triggered_planet_modifier = {
		potential = {
			rse_bio_trophy_effects_active = yes
			owner? = {
				any_owned_starbase = { has_starbase_building = rse_underground_studio }
			}
		}
		robotic_pop_bonus_workforce_mult = @rse_bio_trophy_robotic_efficiency
	}
	triggered_planet_modifier = {
		potential = {
			rse_bio_trophy_effects_active = yes
			owner? = {
				has_technology = tech_rse_service_specialization
			}
		}
		planet_stability_add = @rse_bio_trophy_stability
	}
	triggered_planet_modifier = {
		potential = {
			rse_bio_trophy_effects_active = yes
			owner? = { has_policy_flag = rse_activity_scope_urban }
		}
		pop_cat_complex_drone_bonus_workforce_mult = @rse_rotation_category_minor
		worker_and_simple_drone_cat_bonus_workforce_mult = @rse_rotation_category_minor
	}
	triggered_planet_modifier = {
		potential = {
			rse_bio_trophy_effects_active = yes
			owner? = { has_policy_flag = rse_activity_scope_unrestricted }
		}
		pop_cat_complex_drone_bonus_workforce_mult = @rse_rotation_category_balanced
		worker_and_simple_drone_cat_bonus_workforce_mult = @rse_rotation_category_balanced
	}
	triggered_planet_modifier = {
		potential = {
			rse_bio_trophy_effects_active = yes
			owner? = { has_policy_flag = rse_activity_scope_rural }
		}
		pop_cat_complex_drone_bonus_workforce_mult = @rse_rotation_category_minor
		worker_and_simple_drone_cat_bonus_workforce_mult = @rse_rotation_category_major
		planet_orbital_bombardment_damage = @rse_rotation_rural_bombardment
	}
	triggered_planet_modifier = {
		potential = {
			rse_bio_trophy_effects_active = yes
			owner? = { has_technology = tech_rse_service_specialization }
		}
		# Tuned after in-game pop-group testing: ten service jobs per Bio-Trophy
		# job unit keeps the supporting workforce strong without overwhelming
		# every other complex-drone job.
		job_rse_home_service_add = 10
	}
	triggered_planet_modifier = {
		potential = {
			rse_bio_trophy_effects_active = yes
			owner? = { has_technology = tech_rse_personal_records }
		}
		job_rse_activity_recorder_add = 10
	}
	weight = { weight = @special_job_weight }
	inline_script = "jobs/automodding_priority_charisma"
}

# Dedicated replacement used only by the unmonitored "Exhaust Them" plan.
# It is uncapped like vanilla organic_exhibit: eligibility is mutually
# exclusive with bio_trophy, so pop groups move without changing citizenship,
# population upkeep or the capacity definitions supplied by other content.
rse_exhausted_bio_trophy = {
	category = bio_trophy
	can_be_automated = no
	can_set_priority = no
	is_capped_by_modifier = no
	swappable_data = {
		default = {
			building_icon = building_organic_sanctuary
			icon = bio_trophy
		}
	}
	possible_pre_triggers = {
		is_sapient = yes
		is_robotic = no
	}
	possible = {
		has_citizenship_type = { type = citizenship_organic_trophy country = owner }
		OR = {
			is_pop_category = bio_trophy
			is_pop_category = bio_trophy_unemployment
		}
		colony? = {
			NOT = { is_planet_class = pc_cosmogenesis_world }
			NOT = { has_carrier_flag = rse_growth_monitor_enabled }
		}
		owner? = {
			has_country_flag = rse_route_exclusive_service
			has_policy_flag = rse_growth_exhaust
			is_hospitality_servitor_empire = no
		}
		has_trait = trait_rse_protected_founder
	}
	resources = { category = planet_rse_exhausted_bio_trophies }
	weight = { weight = @special_job_weight }
}
