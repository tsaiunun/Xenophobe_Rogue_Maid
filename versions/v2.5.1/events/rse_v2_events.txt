namespace = rse_v2

# Monthly country state machine. Counters are integers in months so the events
# survive saving, loading and hot updates without relying on delayed scopes.
country_event = {
	id = rse_v2.1
	hide_window = yes
	is_triggered_only = yes

	trigger = { rse_has_exclusive_route = yes }

	immediate = {
		if = {
			limit = { has_ascension_perk = ap_rse_master_maid }
			remove_modifier = rse_agenda_expand_service
		}

		# The tradition finisher contributes the same +5 monthly progress as the
		# vanilla Transformation situation's base value, without overriding it.
		if = {
			limit = { has_tradition = tr_rse_exclusive_service_finish }
			every_situation = {
				limit = {
					is_situation_type = situation_transformation
					NOT = { has_situation_flag = situation_transformation_pause }
				}
				add_situation_progress = 5
			}
		}

		# Hot-update migration for saves that completed the event before the
		# first Bio-Trophy leader and the separate recruitment unlock existed.
		if = {
			limit = {
				has_country_flag = rse_great_species_completed
				NOT = { has_country_flag = rse_trophy_leader_recruitment_unlocked }
			}
			country_event = { id = rse_v2.51 }
		}

		# Consciousness Control is visible on every protected organic leader once
		# its technology is known. The death replacement on_action keys from this
		# trait instead of silently applying to unrelated leaders.
		if = {
			limit = { has_technology = tech_rse_mind_control }
				every_owned_leader = {
				limit = {
					is_robotic_species = no
					has_trait = leader_trait_rse_personal_service
					NOT = { has_trait = leader_trait_rse_mind_control }
				}
				add_trait = { trait = leader_trait_rse_mind_control }
			}
		}

		if = {
			limit = {
				NOT = { has_country_flag = rse_mugwort_completed }
				any_owned_starbase = { has_starbase_building = rse_underground_studio }
			}
			change_variable = { which = rse_mugwort_months value = 1 }
			if = {
				limit = { check_variable = { which = rse_mugwort_months value >= 12 } }
				country_event = { id = rse_v2.40 }
			}
		}
		else = {
			clear_variable = rse_mugwort_months
		}

		if = {
			limit = {
				has_technology = tech_rse_social_simulation
				NOT = { has_country_flag = rse_great_species_completed }
			}
			change_variable = { which = rse_great_species_months value = 1 }
			if = {
				limit = { check_variable = { which = rse_great_species_months value >= 12 } }
				country_event = { id = rse_v2.20 }
			}
		}

		if = {
			limit = {
				has_policy_flag = rse_growth_all_time
				NOT = { has_country_flag = rse_maid_runaway_resolved }
			}
			change_variable = { which = rse_growth_all_time_months value = 1 }
			if = {
				limit = { check_variable = { which = rse_growth_all_time_months value >= 12 } }
				country_event = { id = rse_v2.30 }
			}
		}
		else = {
			set_variable = { which = rse_growth_all_time_months value = 0 }
		}
	}
}

country_event = {
	id = rse_v2.2
	hide_window = yes
	is_triggered_only = yes

	trigger = { rse_has_exclusive_route = yes }

	immediate = {
		if = {
			limit = {
				last_increased_tech = tech_rse_mechanical_maids
				NOT = { has_country_flag = rse_maid_policy_initialized_v2 }
			}
			set_policy = {
				policy = rse_maid_production_policy
				option = rse_maid_production_0
				cooldown = no
			}
			set_country_flag = rse_maid_policy_initialized_v2
		}
		if = {
			limit = {
				last_increased_tech = tech_rse_drone_rotation
				NOT = { has_country_flag = rse_activity_scope_initialized_v2 }
			}
			set_policy = {
				policy = rse_activity_scope_policy
				option = rse_activity_scope_urban
				cooldown = no
			}
			set_country_flag = rse_activity_scope_initialized_v2
		}
		if = {
			limit = {
				last_increased_tech = tech_rse_desire_analysis
				NOT = { has_country_flag = rse_growth_policy_initialized_v2 }
			}
			set_policy = {
				policy = rse_organic_growth_policy
				option = rse_growth_maintain
				cooldown = no
			}
			set_country_flag = rse_growth_policy_initialized_v2
		}
	}
}

# Create and mark the first Bio-Trophy leader. New games call this from
# "A Great Species Once More"; the monthly migration also repairs older saves.
country_event = {
	id = rse_v2.51
	hide_window = yes
	is_triggered_only = yes

	immediate = {
		set_country_flag = rse_trophy_leader_recruitment_unlocked
		if = {
			limit = { NOT = { any_owned_leader = { has_leader_flag = rse_first_trophy_leader } } }
			if = {
				limit = {
					any_owned_leader = {
						leader_class = scientist
						is_robotic_species = no
						has_trait = leader_trait_rse_personal_service
					}
				}
				random_owned_leader = {
					limit = {
						leader_class = scientist
						is_robotic_species = no
						has_trait = leader_trait_rse_personal_service
					}
					set_leader_flag = rse_first_trophy_leader
				}
			}
			else = {
				random_owned_species = {
					limit = { has_trait = trait_rse_protected_founder }
					save_event_target_as = rse_first_trophy_species
				}
				if = {
					limit = { exists = event_target:rse_first_trophy_species }
					create_leader = {
						class = scientist
						species = event_target:rse_first_trophy_species
						name = random
						skill = 3
						event_leader = yes
					}
					last_created_leader = {
						set_leader_flag = rse_first_trophy_leader
						add_trait = { trait = leader_trait_rse_personal_service }
						if = {
							limit = { owner = { has_country_flag = rse_maid_runaway_accepted } }
							add_trait = { trait = leader_trait_rse_regular_rest }
						}
					}
				}
			}
		}
	}
}
